function formatPrice(value) {
  const price = Number(value);

  return Number.isFinite(price)
    ? `ƒ,1${price.toLocaleString("en-IN", {
        maximumFractionDigits: 2,
      })}`
    : "Price unavailable";
}

function extractItems(responseData) {
  if (Array.isArray(responseData)) return responseData;
  if (Array.isArray(responseData?.items)) return responseData.items;
  if (Array.isArray(responseData?.content)) return responseData.content;
  return [];
}

function getWishlistProductId(item) {
  return item?.productId ?? item?.product?.id ?? item?.id;
}

export default function HomePage() {
  const { user, loading: authLoading } = useAuth();
  const navigate = useNavigate();

  const [products, setProducts] = useState([]);
  const [productsLoading, setProductsLoading] = useState(true);
  const [productsError, setProductsError] = useState("");

  const [categoriesFromApi, setCategoriesFromApi] = useState([]);
  const [search, setSearch] = useState("");
  const [activeCategory, setActiveCategory] = useState("All");

  const [wishlist, setWishlist] = useState([]);
  const [wishlistLoading, setWishlistLoading] = useState(false);
  const [wishlistError, setWishlistError] = useState("");
  const [wishlistActionId, setWishlistActionId] = useState(null);
  const [showWishlist, setShowWishlist] = useState(false);

  // Load products.
  useEffect(() => {
    let cancelled = false;

    async function loadProducts() {
      setProductsLoading(true);
      setProductsError("");

      try {
        const data = await productService.getProducts();

        if (!cancelled) {
          setProducts(extractItems(data));
        }
      } catch (error) {
        console.error("Failed to load products:", error);

        if (!cancelled) {
          setProductsError(
            "We couldn't load products right now. Please try again."
          );
        }
      } finally {
        if (!cancelled) {
          setProductsLoading(false);
        }
      }
    }

    loadProducts();

    return () => {
      cancelled = true;
    };
  }, []);

  // Load categories.
  useEffect(() => {
    let cancelled = false;

    async function loadCategories() {
      try {
        const data = await productService.getCategories();

        if (cancelled) return;

        if (Array.isArray(data)) {
          setCategoriesFromApi(
            data
              .map((category) =>
                typeof category === "string"
                  ? category
                  : category?.name
              )
              .filter(Boolean)
          );
        }
      } catch (error) {
        console.error("Failed to load categories:", error);
      }
    }

    loadCategories();

    return () => {
      cancelled = true;
    };
  }, []);

  // Load the authenticated user's wishlist.
  useEffect(() => {
    let cancelled = false;

    async function loadWishlist() {
      if (authLoading) return;

      if (!user) {
        setWishlist([]);
        setWishlistError("");
        setWishlistLoading(false);
        return;
      }

      setWishlistLoading(true);
      setWishlistError("");

      try {
        const response = await api.get("/wishlist");

        if (!cancelled) {
          setWishlist(extractItems(response.data));
        }
      } catch (error) {
        console.error("Failed to load Wishlist:", error);

        if (!cancelled) {
          setWishlistError(
            "Unable to load your Wishlist. Please try again."
          );
        }
      } finally {
        if (!cancelled) {
          setWishlistLoading(false);
        }
      }

    loadWishlist();

    return () => {
      cancelled = true;
    };
  }, [user, authLoading]);

  const categories = useMemo(() => {

      return categoriesFromApi;

      ),

      const matchesCategory =
        activeCategory === "All" ||
        product.category === activeCategory;

      const searchable = [
        product.name,
        product.description,
        product.category,

        .filter(Boolean)
        .join(" ")
        .toLowerCase();

  function isInWishlist(productId) {
    return wishlist.some(
      (item) =>
        String(getWishlistProductId(item)) === String(productId)

  async function toggleWishlist(product) {
    if (!user) {
      navigate("/login", {
        state: { from: { pathname: "/" } },

      return;
    }

    const productId = product?.id;

    if (productId == null) {
      setWishlistError("This product does not have a valid ID.");
      return;
    }

    if (wishlistActionId !== null) return;

    const alreadyWished = isInWishlist(productId);

    setWishlistActionId(String(productId));
    setWishlistError("");

    try {
      if (alreadyWished) {
        await api.delete(`/wishlist/items/${productId}`);

        setWishlist((current) =>
          current.filter(
            (item) =>
              String(getWishlistProductId(item)) !==
              String(productId)
          )
        );
      } else {
        const response = await api.post(
          `/wishlist/items/${productId}`
        );

        const returnedItem = response?.data;

        const newItem =
          returnedItem &&
          typeof returnedItem === "object" &&
          (
            returnedItem.productId != null ||
            returnedItem.product?.id != null
          )
            ? returnedItem
            : {
                productId: product.id,
                name: product.name,
                description: product.description,
                category: product.category,
                imageUrl: product.imageUrl,
                price: product.price,
                discountedPrice: product.discountedPrice,
                stockQuantity: product.stockQuantity,
              };

        setWishlist((current) => {
          const exists = current.some(
            (item) =>
              String(getWishlistProductId(item)) ===
              String(productId)
          );

          return exists ? current : [...current, newItem];
        });
      }
    } catch (error) {
      console.error("Wishlist update failed:", error);

      setWishlistError(
        error?.response?.data?.message ||
          "Could not update your Wishlist. Please try again."
      );
    } finally {
      setWishlistActionId(null);

        item.imageUrl ??

        catalogueProduct?.imageUrl,
      price:
        item.price ??
        item.product?.price ??
        catalogueProduct?.price,
      description:
        item.description ??
        item.product?.description ??
        catalogueProduct?.description,

        <div className="relative mx-auto grid max-w-[1440px] items-center gap-10 px-5 py-16 sm:py-20 lg:grid-cols-2 lg:px-10 lg:py-24">
          <div>

  const [products, setProducts] = useState([]);

  const [wishlistError, setWishlistError] = useState("");

  const [error, setError] = useState("");

  const [search, setSearch] = useState("");
  const [selectedCategory, setSelectedCategory] = useState("");
  const [selectedSubcategory, setSelectedSubcategory] = useState("");
  const [minPrice, setMinPrice] = useState("");
  const [maxPrice, setMaxPrice] = useState("");
  const [stockOnly, setStockOnly] = useState(false);

  const [sortBy, setSortBy] = useState("id");
  const [direction, setDirection] = useState("asc");
  const [page, setPage] = useState(0);
  const [retryKey, setRetryKey] = useState(0);

  const [pageData, setPageData] = useState({
    totalElements: 0,
    totalPages: 0,
    number: 0,
  });

  const { user, loading: authLoading } = useAuth();
  const navigate = useNavigate();

  const visibleCategories = useMemo(
    () => ["", ...categories],
    [categories]
  );

  // Load categories.
  useEffect(() => {
    let cancelled = false;

    async function loadCategories() {
      try {
        const data = await productService.getCategories();

        if (!cancelled) {
          setCategories(Array.isArray(data) ? data : []);
        }
      } catch {
        if (!cancelled) {
          setCategories([]);
        }
      }
    }

    loadCategories();

    return () => {
      cancelled = true;
    };
  }, []);

  // Load subcategories when the selected category changes.
  useEffect(() => {
    let cancelled = false;

    async function loadSubcategories() {
      setSelectedSubcategory("");
      setSubcategories([]);

      if (!selectedCategory) {
        setCategoryLoading(false);
        return;
      }

      setCategoryLoading(true);

      try {
        const data =
          await productService.getSubcategories(selectedCategory);

        if (!cancelled) {
          setSubcategories(Array.isArray(data) ? data : []);
        }
      } catch {
        if (!cancelled) {
          setSubcategories([]);
        }
      } finally {
        if (!cancelled) {
          setCategoryLoading(false);
        }
      }
    }

    loadSubcategories();

    return () => {
      cancelled = true;
    };
  }, [selectedCategory]);

  // Load products using backend search, filters, sorting and pagination.
  useEffect(() => {
    let cancelled = false;

    const timer = setTimeout(async () => {
      setLoading(true);
      setError("");

      try {
        const result = await productService.searchProducts({
          keyword: search,
          category: selectedCategory,
          subcategory: selectedSubcategory,
          minPrice,
          maxPrice,
          inStock: stockOnly,
          page,
          size: PAGE_SIZE,
          sortBy,
          direction,
        });

        if (cancelled) return;

        setProducts(
          Array.isArray(result?.content) ? result.content : []
        );

        setPageData({
          totalElements: result?.totalElements ?? 0,
          totalPages: result?.totalPages ?? 0,
          number: result?.number ?? page,
        });
      } catch (err) {
        if (cancelled) return;

        setProducts([]);
        setPageData({
          totalElements: 0,
          totalPages: 0,
          number: 0,
        });

        setError(
          err.response?.data?.detail ||
            err.response?.data?.message ||
            "Unable to load products. Please check your connection and try again."
        );
      } finally {
        if (!cancelled) {
          setLoading(false);
        }
      }
    }, 300);

    return () => {
      cancelled = true;
      clearTimeout(timer);
    };
  }, [
    search,
    selectedCategory,
    selectedSubcategory,
    minPrice,
    maxPrice,
    stockOnly,
    sortBy,
    direction,

    retryKey,

      if (!user) {
        setWishlistIds(new Set());
        setWishlistError("");
        setWishlistLoading(false);
        return;
      }

      setWishlistLoading(true);

      try {
        const response = await api.get("/wishlist");

        if (cancelled) return;

        const data = response.data;
        const items = Array.isArray(data)
          ? data
          : Array.isArray(data?.items)
            ? data.items
            : Array.isArray(data?.content)
              ? data.content
              : Array.isArray(data?.products)
                ? data.products
                : [];

        const ids = items
          .map((item) => {
            const id =
              item?.productId ??
              item?.product?.id ??
              item?.id;

            return id == null ? null : String(id);
          })
          .filter((id) => id !== null);

        setWishlistIds(new Set(ids));
      } catch (err) {
        if (cancelled) return;

        setWishlistError(
          err.response?.data?.detail ||
            err.response?.data?.message ||
            "Could not load your wishlist. Please retry."
        );
      } finally {
        if (!cancelled) {
          setWishlistLoading(false);
        }
      }

    return () => {
      cancelled = true;

  function getDiscount(product) {
    const value = Number(product.discountPercentage ?? 0);

    if (!Number.isFinite(value)) return 0;

    return Math.min(100, Math.max(0, value));
  }

  function getFinalPrice(product) {
    const price = Number(product.price || 0);
    const discount = getDiscount(product);

    return price * (1 - discount / 100);
  }

  function getErrorMessage(err, fallback) {
    return (

      err.response?.data?.message ||

      setAddingId(null);

          const updated = new Set(currentIds);
          updated.add(productId);
          return updated;
        });

        setMessage(`${product.name} added to your wishlist.`);
      }
    } catch (err) {
      setWishlistError(
        getErrorMessage(
          err,
          "Could not update your wishlist. Please try again."
        )
      );
    } finally {
      setWishlistActionId(null);
    }

import { useEffect, useState } from "react";
import { Link, useNavigate, useParams } from "react-router-dom";
import {
  ArrowLeft,
  Minus,
  Plus,
  Package,
  ShoppingCart,
  CheckCircle2,
} from "lucide-react";

import WishlistButton from "../components/common/WishlistButton";
import productService from "../services/productService";
import cartService from "../services/cartService";
import { useAuth } from "../context/useAuth.js";

export default function ProductDetailsPage() {
  const { id } = useParams();
  const navigate = useNavigate();
  const { user, loading: authLoading } = useAuth();

  const [product, setProduct] = useState(null);
  const [quantity, setQuantity] = useState(1);
  const [loading, setLoading] = useState(true);
  const [adding, setAdding] = useState(false);
  const [error, setError] = useState("");
  const [message, setMessage] = useState("");

  useEffect(() => {
    let cancelled = false;

    async function loadProduct() {
      setLoading(true);
      setError("");
      setProduct(null);
      setMessage("");

      try {
        const data = await productService.getProduct(id);

        if (!cancelled) {
          setProduct(data);
          setQuantity(1);
        }
      } catch (err) {
        if (!cancelled) {
          setError(
            err.response?.status === 404
              ? "This product was not found or is no longer available."
              : "Unable to load this product. Please try again."
          );
        }
      } finally {
        if (!cancelled) {
          setLoading(false);
        }
      }
    }

    loadProduct();

    return () => {
      cancelled = true;
    };
  }, [id]);

  const money = (amount) =>
    `ƒ,1${Number(amount || 0).toLocaleString("en-IN", {
      minimumFractionDigits: 2,
      maximumFractionDigits: 2,
    })}`;

  async function handleAddToCart() {
    if (!product || adding) return;

    setMessage("");
    setError("");

    if (!user) {
      navigate("/login", {
        state: {
          from: {
            pathname: `/products/${id}`,
          },
        },
      });
      return;
    }

    setAdding(true);

    try {
      await cartService.addItem(product.id, quantity);
      setMessage(`${product.name} added to your cart.`);
    } catch (err) {
      setError(
        err.response?.data?.detail ||
          err.response?.data?.message ||
          "Unable to add this product to your cart."
      );
    } finally {
      setAdding(false);
    }
  }

  const stockQuantity = Number(product?.stockQuantity ?? 0);

  return (
    <div className="min-h-screen bg-slate-950 text-white">
      {/* Shared StorefrontLayout renders the common header and footer. */}

      <section className="mx-auto max-w-7xl px-4 py-8 sm:px-6 lg:px-8">
        <Link
          to="/products"
          className="inline-flex items-center gap-2 text-sm font-medium text-slate-400 transition hover:text-violet-300"
        >
          <ArrowLeft size={17} />
          Back to products
        </Link>

        {loading ? (
          <div className="mt-8 grid gap-8 lg:grid-cols-2">
            <div className="h-96 animate-pulse rounded-3xl bg-white/5" />
            <div className="h-96 animate-pulse rounded-3xl bg-white/5" />
          </div>
        ) : error && !product ? (
          <div
            role="alert"
            className="mt-8 rounded-2xl border border-red-400/20 bg-red-400/10 p-6 text-red-200"
          >
            {error}

            <div className="mt-5">
              <Link
                to="/products"
                className="font-semibold text-violet-300 underline"
              >
                Return to products
              </Link>
            </div>
          </div>
        ) : product ? (
          <div className="mt-8 grid items-start gap-8 lg:grid-cols-2 lg:gap-14">
            {/* Product image and wishlist button */}
            <div className="relative">
              <div className="flex min-h-80 items-center justify-center overflow-hidden rounded-3xl border border-white/10 bg-gradient-to-br from-violet-500/10 via-slate-900 to-indigo-500/10 sm:min-h-[480px]">
                {product.imageUrl ? (
                  <img
                    src={product.imageUrl}
                    alt={product.name}
                    onError={(event) => {
                      event.currentTarget.style.display = "none";
                    }}
                    className="h-full max-h-[560px] w-full object-contain"
                  />
                ) : (
                  <Package
                    size={100}
                    className="text-violet-300/50"
                  />
                )}
              </div>

              <WishlistButton
                product={product}
                className="absolute right-4 top-4 z-10"
              />
            </div>

            {/* Product details */}
            <div className="py-2 sm:py-5">
              <p className="text-xs font-bold uppercase tracking-[0.2em] text-violet-300">
                {product.category || "ASTRA Collection"}
              </p>

              <h1 className="mt-4 text-3xl font-extrabold tracking-tight sm:text-4xl">
                {product.name}
              </h1>

              <p className="mt-5 text-3xl font-bold text-violet-200">
                {money(product.price)}
              </p>

              {product.sku && (
                <p className="mt-2 text-sm text-slate-400">
                  SKU: {product.sku}
                </p>
              )}

                <p className="mt-2 text-sm text-slate-400">
                  Subcategory: {product.subcategory}

                    "Discover quality and convenience with this ASTRA product."}

                              variant.color,
                            ])
                            .filter(Boolean)
                        ),

function normalizeStatus(value) {
  const status = String(value || "PLACED")
    .trim()
    .toUpperCase()
    .replace(/[\s-]+/g, "_");

  if (status === "ORDER_PLACED") return "PLACED";
  if (status === "PROCESSING") return "CONFIRMED";

  return status;
}

function formatPrice(value) {
  return new Intl.NumberFormat("en-IN", {
    style: "currency",
    currency: "INR",
    maximumFractionDigits: 2,
  }).format(Number(value || 0));
}

function formatDate(value) {
  if (!value) return "Not available";

  const date = new Date(value);

  if (Number.isNaN(date.getTime())) return "Not available";

  return date.toLocaleString("en-IN", {
    day: "2-digit",
    month: "short",
    year: "numeric",
    hour: "2-digit",
    minute: "2-digit",
  });
}

function getErrorMessage(err) {
  const responseStatus = err.response?.status;

  if (responseStatus === 401) {
    return "Session expired. Please log in again.";
  }

  if (responseStatus === 403) {
    return "You do not have permission to view this order.";
  }

  if (responseStatus === 404) {
    return "Order not found or you do not have access to it.";
  }

  return (
    err.response?.data?.detail ||
    err.response?.data?.message ||
    "Unable to load order tracking. Please try again."
  );
}

function getPasswordRules(password) {
  return [
    {
      label: "At least 8 characters",
      valid: password.length >= 8 && password.length <= 72,
    },
    {
      label: "One uppercase letter",
      valid: /[A-Z]/.test(password),
    },
    {
      label: "One lowercase letter",
      valid: /[a-z]/.test(password),
    },
    {
      label: "One number",
      valid: /\d/.test(password),
    },
    {
      label: "One special character",
      valid: /[^A-Za-z0-9]/.test(password),
    },
  ];
}

function passwordIsStrong(password) {
  return getPasswordRules(password).every((rule) => rule.valid);
}

function calculateAge(dateOfBirth) {
  if (!dateOfBirth) return "";

  const dob = new Date(`${dateOfBirth}T00:00:00`);

  if (Number.isNaN(dob.getTime())) return "";

  const today = new Date();

  let age = today.getFullYear() - dob.getFullYear();

  if (
    today.getMonth() < dob.getMonth() ||
    (today.getMonth() === dob.getMonth() &&
      today.getDate() < dob.getDate())
  ) {
    age--;
  }

  return age >= 0 ? age : "";
}

function formatDate(value) {
  if (!value) return "Not available";

  const date = new Date(value);

  if (Number.isNaN(date.getTime())) return "Not available";

  return date.toLocaleDateString("en-IN", {
    day: "2-digit",
    month: "short",
    year: "numeric",
  });
}

import { useState } from "react";
import authService from "../services/authService";

export default function ChangePasswordForm() {
  const [form, setForm] = useState({
    currentPassword: "",
    newPassword: "",
    confirmPassword: "",
  });

  const [loading, setLoading] = useState(false);
  const [error, setError] = useState("");
  const [success, setSuccess] = useState("");

  const handleChange = (event) => {
    const { name, value } = event.target;

    setForm((previous) => ({
      ...previous,
      [name]: value,
    }));

    setError("");
    setSuccess("");
  };

  const handleSubmit = async (event) => {
    event.preventDefault();
    setError("");
    setSuccess("");

    if (form.newPassword.length < 8) {
      setError("New password must contain at least 8 characters.");
      return;
    }

    if (form.newPassword.length > 72) {
      setError("New password cannot exceed 72 characters.");
      return;
    }

    if (form.newPassword !== form.confirmPassword) {
      setError("New password and confirmation do not match.");
      return;
    }

    if (form.currentPassword === form.newPassword) {
      setError("Choose a password different from your current password.");
      return;
    }

    try {
      setLoading(true);

      const response = await authService.changePassword(
        form.currentPassword,
        form.newPassword
      );

      setSuccess(response.message || "Password changed successfully.");

      setForm({
        currentPassword: "",
        newPassword: "",
        confirmPassword: "",
      });
    } catch (err) {
      setError(
        err.response?.data?.detail ||
        err.response?.data?.message ||
        (typeof err.response?.data === "string"
          ? err.response.data
          : "") ||
        "Unable to change password. Please check your current password and try again."
      );
    } finally {
      setLoading(false);
    }
  };

  const inputClass =
    "mt-2 w-full rounded-xl border border-white/10 bg-white/5 px-4 py-3 text-white outline-none transition placeholder:text-gray-500 focus:border-violet-400 focus:ring-2 focus:ring-violet-500/20";

  return (
    <section className="rounded-2xl border border-white/10 bg-[#171326] p-5 text-white shadow-xl sm:p-7">
      <div className="mb-6">
        <p className="text-sm font-medium text-violet-300">
          SECURITY SETTINGS
        </p>
        <h2 className="mt-2 text-2xl font-bold">
          Change password
        </h2>
        <p className="mt-2 text-sm text-gray-400">
          Enter your current password and choose a new, secure password.
        </p>
      </div>

      {error && (
        <div
          role="alert"
          className="mb-4 rounded-xl border border-red-500/30 bg-red-500/10 p-3 text-sm text-red-300"
        >
          {error}
        </div>
      )}

      {success && (
        <div
          role="status"
          className="mb-4 rounded-xl border border-emerald-500/30 bg-emerald-500/10 p-3 text-sm text-emerald-300"
        >
          {success}
        </div>
      )}

      <form onSubmit={handleSubmit} className="space-y-5">
        <div>
          <label
            htmlFor="currentPassword"
            className="text-sm font-medium text-gray-200"
          >
            Current password
          </label>
          <input
            id="currentPassword"
            name="currentPassword"
            type="password"
            autoComplete="current-password"
            value={form.currentPassword}
            onChange={handleChange}
            className={inputClass}
            placeholder="Enter current password"
            required
          />
        </div>

        <div>
          <label
            htmlFor="newPassword"
            className="text-sm font-medium text-gray-200"
          >
            New password
          </label>
          <input
            id="newPassword"
            name="newPassword"
            type="password"
            autoComplete="new-password"
            minLength={8}
            maxLength={72}
            value={form.newPassword}
            onChange={handleChange}
            className={inputClass}
            placeholder="At least 8 characters"
            required
          />
        </div>

        <div>
          <label
            htmlFor="confirmPassword"
            className="text-sm font-medium text-gray-200"
          >
            Confirm new password
          </label>
          <input
            id="confirmPassword"
            name="confirmPassword"
            type="password"
            autoComplete="new-password"
            minLength={8}
            maxLength={72}
            value={form.confirmPassword}
            onChange={handleChange}
            className={inputClass}
            placeholder="Re-enter new password"
            required
          />
        </div>

        <button
          type="submit"
          disabled={loading}
          className="w-full rounded-xl bg-violet-600 px-5 py-3 font-semibold text-white transition hover:bg-violet-500 disabled:cursor-not-allowed disabled:opacity-60 sm:w-auto"
        >
          {loading ? "Updating password..." : "Update password"}
        </button>
      </form>
    </section>
  );
}
